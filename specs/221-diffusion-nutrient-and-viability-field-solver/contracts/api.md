# Contract: public API (R02-F21)

VisualCoreSimulation (pure). Throwing members use `throws(NutrientError)`.
- `public enum NutrientError: Error, Equatable, Sendable { case gridSize, channel(Int), cellRange(Int), amountLimit, parameter(String), time, sourceLimit, growth(String), newerSchema(Int), malformed(String); public var reasonCode: String }`
- `public enum NutrientSourceKind: String, Sendable, Codable, CaseIterable { case source, sink }`
- `public struct NutrientSource: Sendable, Equatable { public let cell: Int; public let channel: Int; public let kind: NutrientSourceKind; public let rate: Double; public internal(set) var carry: Double; public init(cell: Int, channel: Int, kind: NutrientSourceKind, rate: Double) throws }` (cell and channel are checked when added to a field)
- `public struct ViabilityBand: Sendable, Equatable, Codable { public let low, high, ramp: Double; public init(low: Double, high: Double, ramp: Double = 0) throws; public func membership(_ units: Double) -> Double }`
- `public struct NutrientAccount: Sendable, Equatable, Codable { public var added, removed, faded, eaten: Int64; public var net: Int64 }`
- `public struct NutrientLedger: Sendable, Equatable, Codable { public var accounts: [NutrientAccount] }`
- `public struct NutrientStepReport: Sendable, Equatable { public let substeps: Int; public let simulated, dropped: Double; public let added, removed, faded, eaten: [Int64]; public let capped, short: Int64; public let grown: Int }`
- `public struct GrowthNetwork: Sendable, Equatable { public static let maxCells = 65_536; public static let maxPerFrame = 1_024; public let limit, perFrame: Int; public let threshold: Double; public let foodChannel: Int; public let cost: Int64; public internal(set) var cells: [Int]; public internal(set) var parents: [Int] /* −1 for a seed */; public init(limit: Int, perFrame: Int, threshold: Double, foodChannel: Int, cost: Int64) throws; public func contains(_ cell: Int) -> Bool }`
- `public struct NutrientField: Sendable, Equatable, Codable { public static let quantaPerUnit: Int64 = 65_536; maxAmount: Int64 = 1 << 40; minSide = 2; maxSide = 512; maxCells = 262_144; maxChannels = 4; maxDiffusion = 1_000.0; maxDecay = 100.0; maxFrame = 0.25; maxSubsteps = 256; maxSources = 256; public let width, height, channels: Int; public let diffusion, decay: [Double]; public private(set) var sources: [NutrientSource]; public private(set) var bands: [ViabilityBand?]; public private(set) var network: GrowthNetwork?; public private(set) var ledger: NutrientLedger; public init(width: Int, height: Int, channels: Int = 1, diffusion: [Double], decay: [Double]? = nil /* zeros */) throws; public var cellCount: Int; public func cell(x: Int, y: Int) -> Int; public func amount(_ channel: Int, at cell: Int) -> Int64 /* 0 out of range */; public func total(_ channel: Int) -> Int64; public var isBalanced: Bool; public mutating func add(_ quanta: Int64, channel: Int, at cell: Int) throws; public mutating func remove(_ quanta: Int64, channel: Int, at cell: Int) throws -> Int64 /* takes at most what the cell holds, counted as removed */; public mutating func addSource(_ source: NutrientSource) throws; public mutating func setBand(_ band: ViabilityBand?, channel: Int) throws; public func viability(at cell: Int) -> Double; public mutating func setNetwork(_ network: GrowthNetwork?) throws /* .growth("foodChannel") out of range; replaces any network */; public mutating func seed(at cell: Int) throws /* .growth("network"), .cellRange(c), .growth("seed") on a network cell, .growth("limit") */; public mutating func step(by time: Double) throws -> NutrientStepReport; public func fieldGrid(channel: Int) -> FieldGrid?; public func viabilityGrid() -> FieldGrid; public func networkGraph() -> RouteGraph; public func encoded() -> Data; public static func decode(_ data: Data) throws -> NutrientField }`
