# Contract: public API (R02-F19)

VisualCoreScenes (pure). Throwing members use `throws(PlantError)` unless they wrap `MeshError`.
- `public enum PlantError: Error, Equatable, Sendable { case generations(Int), wordLimit(Int), rule(String), weight(String), brackets, turtleParameter(String), shapeParameter(String), branchLimit(Int), scatterArea, scatterParameter(String), newerSchema(Int), malformed(String); public var reasonCode: String }`
- `public struct PlantRule: Sendable, Equatable, Codable { public let predecessor: Character; public let successors: [(word: String, weight: Double)] /* stored as an array of objects */; public init(_ predecessor: Character, _ word: String) throws; public init(_ predecessor: Character, weighted: [(String, Double)]) throws }` (Equatable and Codable written by hand because of the tuples.)
- `public struct PlantGrammar: Sendable, Equatable, Codable { public static let schema = 1; public static let maxGenerations = 12, maxWord = 1_048_576, maxRules = 64, maxBranches = 65_536; public let start: String; public let rules: [PlantRule]; public let generations: Int; public let angle, step, radius, radiusFactor: Double; public let seed: UInt64; public init(start: String, rules: [PlantRule], generations: Int, angle: Double, step: Double, radius: Double = 0.05, radiusFactor: Double = 0.8, seed: UInt64 = 0) throws; public func encoded() -> Data; public static func decode(_ data: Data) throws(PlantError) -> PlantGrammar }`
- `public enum LSystem { public static func derive(_ grammar: PlantGrammar) throws(PlantError) -> String }`
- `public struct PlantBranch: Sendable, Equatable { public let start, end: Vec3; public let radius: Double; public let depth: Int }`
- `public enum PlantPart: String, Sendable, Codable, CaseIterable { case petal, cap, leaf }`
- `public struct PlantMarker: Sendable, Equatable { public let kind: PlantPart; public let position, heading, left, up: Vec3; public let size: Double }`
- `public struct PlantSkeleton: Sendable, Equatable { public let branches: [PlantBranch]; public let markers: [PlantMarker] }`
- `public enum PlantTurtle { public static func interpret(_ word: String, grammar: PlantGrammar) throws(PlantError) -> PlantSkeleton; public static func grow(_ grammar: PlantGrammar) throws(PlantError) -> PlantSkeleton /* derive then interpret */ }`
- `public struct PetalShape: Sendable, Equatable { public let length, width, curl: Double; public let segments: Int; public init(length:width:curl:segments:) throws(PlantError); public func mesh() throws(MeshError) -> Mesh /* spine along +y from the origin, bending towards +z, width along x */ }`
- `public struct CapShape: Sendable, Equatable { public let radius, height, rim: Double; public let sides, rings: Int; public init(radius:height:rim:sides:rings:) throws(PlantError); public func mesh() throws(MeshError) -> Mesh /* axis +y, base at y = −rim, apex at y = height */ }`
- `public struct StemShape: Sendable, Equatable { public let length, baseRadius, tipRadius, bend: Double; public let sides, segments: Int; public init(length:baseRadius:tipRadius:bend:sides:segments:) throws(PlantError); public func mesh() throws(MeshError) -> Mesh /* from the origin along +y, bending towards +z */ }`
- `public enum PlantMesh { public static func build(_ skeleton: PlantSkeleton, petal: PetalShape, cap: CapShape, budget: MeshBudget = .default) throws(MeshError) -> Mesh }`
- `public struct PlantInstance: Sendable, Equatable { public let position: Vec3; public let scale, yaw: Double; public let variant: Int; public var model: Mat4 }`
- `public enum PlantScatter { public static let maxCount = 65_536; public static func scatter(minX: Double, maxX: Double, minZ: Double, maxZ: Double, spacing: Double, maxCount: Int, scale: ClosedRange<Double>, variants: Int, seed: UInt64) throws(PlantError) -> [PlantInstance] }`
