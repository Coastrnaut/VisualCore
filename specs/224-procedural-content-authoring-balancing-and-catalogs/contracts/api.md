# Contract: public API (R02-F24)

VisualCoreSimulation (pure). Throwing members use `throws(ContentError)`.
- `public enum ContentError: Error, Equatable, Sendable { case plan(String), planFailed(stage: Int, calls: Int), progression(Int), tutorial(String), newerSchema(Int), malformed(String), edited; public var reasonCode: String }`
- `public struct PuzzleDifficulty: Sendable, Equatable, Hashable, Codable, Comparable { public let shortest: Int; public let explored: Int; public init(shortest: Int, explored: Int); public init?(certificate: PuzzleCertificate) /* nil without a shortest length */ }`
- `public struct ProgressionRule: Sendable, Equatable, Codable { public static let maxStep = 64; public static let standard: ProgressionRule /* drop 1, jump 3 */; public let maxDrop, maxJump: Int; public init(maxDrop: Int = 1, maxJump: Int = 3) throws; public func breach(_ difficulties: [PuzzleDifficulty]) -> (index: Int, isDrop: Bool)? /* first neighbour breach */ }`
- `public struct PuzzleCatalogStage: Sendable, Equatable, Codable { public static let maxCount = 500; public let count: Int; public let budget: FairnessBudget; public let uniqueness: PuzzleUniqueness; public let attempts: Int; public init(count: Int, budget: FairnessBudget, uniqueness: PuzzleUniqueness = .none, attempts: Int = 100) throws }`
- `public struct PuzzleCatalogPlan: Sendable, Equatable, Codable { public static let maxStages = 16; public static let maxLevels = 1_000; public let id: String; public let seed: UInt64; public let stages: [PuzzleCatalogStage]; public let rule: ProgressionRule; public init(id: String, seed: UInt64, stages: [PuzzleCatalogStage], rule: ProgressionRule = .standard) throws }`
- `public struct PuzzleCatalogEntry<Rules: PuzzleRules>: Sendable, Equatable { public let level: PuzzleLevel<Rules>; public let difficulty: PuzzleDifficulty }`
- `public struct PuzzleCatalog<Rules: PuzzleRules>: Sendable, Equatable { public static var schema: Int { 1 }; public let id: String; public let entries: [PuzzleCatalogEntry<Rules>]; public let digest: String; public init(id: String, entries: [PuzzleCatalogEntry<Rules>]) throws; public static func digest(id: String, identities: [String]) -> String; public func encoded() -> Data; public static func decode(_ data: Data) throws -> PuzzleCatalog }`
- `public enum PuzzleCatalogBuilder { public static func build<R: PuzzleRules>(_ plan: PuzzleCatalogPlan, rules: R, make: (inout SeededRandom, PuzzleCatalogStage) -> (start: R.State, solution: [R.Move])?) throws -> PuzzleCatalog<R> }`
- `public enum PuzzleCatalogIssue: Sendable, Equatable { case uncertified(index: Int, verdict: PuzzleVerdict), duplicate(index: Int, first: Int), difficultyMismatch(index: Int), drop(index: Int), jump(index: Int), digestMismatch }`
- `public struct PuzzleCatalogReport: Sendable, Equatable { public let issues: [PuzzleCatalogIssue]; public var passes: Bool }`
- `public enum PuzzleCatalogChecker { public static func check<R: PuzzleRules>(_ catalog: PuzzleCatalog<R>, rules: R, rule: ProgressionRule) -> PuzzleCatalogReport }`
- `public enum TutorialCondition: Sendable, Equatable, Codable { case move, moves(Int), solve, solveLevel(String), acknowledge }`
- `public struct TutorialContractStep: Sendable, Equatable, Codable { public let id: String; public let textKey: String; public let condition: TutorialCondition; public init(id: String, textKey: String, condition: TutorialCondition) throws }`
- `public struct TutorialContract: Sendable, Equatable, Codable { public static let maxSteps = 64; public let id: String; public let steps: [TutorialContractStep]; public init(id: String, steps: [TutorialContractStep]) throws }`
- `public enum TutorialEvent: Sendable, Equatable { case moved, solved(String), acknowledged }`
- `public struct TutorialContractProgress: Sendable, Equatable { public static var schema: Int { 1 }; public let contractID: String; public private(set) var step: Int; public private(set) var movesInStep: Int; public private(set) var completed: [String]; public private(set) var skipped: [String]; public private(set) var finished: Bool; public init(contract: TutorialContract); public func currentStep(in contract: TutorialContract) -> TutorialContractStep?; public mutating func apply(_ event: TutorialEvent, in contract: TutorialContract) -> Bool /* advanced */; public mutating func skip(in contract: TutorialContract); public mutating func restart(); public func encoded() -> Data; public static func decode(_ data: Data, for contract: TutorialContract) throws -> TutorialContractProgress }`
