# Contract: public API (R02-F20)

VisualCoreSimulation (pure). Throwing members use `throws(SwarmError)`.
- `public enum SwarmError: Error, Equatable, Sendable { case agentLimit(Int), bounds, step, parameter(String), field(String), rasterSize, newerSchema(Int), malformed(String); public var reasonCode: String }`
- `public struct MagneticPole: Sendable, Equatable, Codable { public let position: Vec2; public let strength, softening: Double; public init(position: Vec2, strength: Double, softening: Double = 0.1) throws }`
- `public struct MagneticField: Sendable, Equatable, Codable { public static let maxPoles = 64; public let uniform: Double; public let poles: [MagneticPole]; public static let none: MagneticField; public init(uniform: Double = 0, poles: [MagneticPole] = []) throws; public func poleForce(at position: Vec2, charge: Double) -> Vec2; public func poleForceBound(charge: Double) -> Double /* Σ 2 |s q| / (3√3 ε²) */; public func turned(_ velocity: Vec2, charge: Double, step: Double) -> Vec2 /* exact rotation by −q B h */ }`
- `public struct NeighbourGrid: Sendable { public static let maxAgents = 65_536; public let cellSize: Double; public init(agents: [SteeringAgent], cellSize: Double) throws(SwarmError) /* .agentLimit(n), .parameter("cellSize") */; public func neighbours(of index: Int, radius: Double, limit: Int) -> [Int] /* equals NeighbourSearch.neighbours(of:in:radius:limit:) for the agents it was built from */ }`
- `public enum SwarmEdges: String, Sendable, Codable, CaseIterable { case walled, wrapping }`
- `public struct Swarm: Sendable, Equatable, Codable { public static let schema = 1; public static let maxAgents = 65_536; public private(set) var agents: [SteeringAgent]; public private(set) var charges: [Double]; public let bounds: SteeringBounds; public let edges: SwarmEdges; public let parameters: SteeringParameters; public let speedLimit: Double; public let field: MagneticField; public init(agents: [SteeringAgent], charges: [Double]? = nil /* zeros */, bounds: SteeringBounds, edges: SwarmEdges = .walled, parameters: SteeringParameters = .init(), speedLimit: Double = 5, field: MagneticField = .none) throws; public mutating func step(by h: Double) throws; public func encoded() -> Data; public static func decode(_ data: Data) throws -> Swarm }` (`SteeringBounds` and `SteeringParameters` are encoded field by field because they are not `Codable`).
- `public struct DensityRaster: Sendable, Equatable { public let width, height: Int; public let bounds: SteeringBounds; public private(set) var cells: [Double] /* row-major, y·width + x */; public init(width: Int, height: Int, bounds: SteeringBounds) throws; public mutating func clear(); public mutating func splat(_ agents: [SteeringAgent]); public static func of(_ swarm: Swarm, width: Int, height: Int) throws -> DensityRaster; public var total: Double; public func fieldGrid() -> FieldGrid /* one channel */ }`
