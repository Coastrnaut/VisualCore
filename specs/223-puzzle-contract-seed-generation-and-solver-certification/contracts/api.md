# Contract: public API (R02-F23)

VisualCoreSimulation (pure). Throwing members use `throws(PuzzleError)`.
- `public enum PuzzleError: Error, Equatable, Sendable { case rulesID, solutionLength, budget(String), attempts, generationFailed(Int), newerSchema(Int), malformed(String), edited; public var reasonCode: String }`
- `public protocol PuzzleRules: Sendable { associatedtype State: Hashable & Codable & Sendable; associatedtype Move: Hashable & Codable & Sendable; var rulesID: String { get }; func moves(in state: State) -> [Move]; func apply(_ move: Move, to state: State) -> State?; func isSolved(_ state: State) -> Bool }`
- `public enum PuzzleUniqueness: String, Sendable, Codable, CaseIterable { case none, goal, solution }`
- `public struct FairnessBudget: Sendable, Equatable, Codable { public static let maxLength = 4_096; public static let maxBranching = 1_024; public static let maxNodes = 1_000_000; public let minLength, maxLength, maxBranching, nodeBudget: Int; public init(minLength: Int = 1, maxLength: Int, maxBranching: Int = 64, nodeBudget: Int = 100_000) throws }`
- `public enum PuzzleVerdict: Sendable, Equatable, Codable { case valid, wrongRules, illegalMove(Int), notSolved, tooShort(Int), tooLong(Int), tooBranching(Int), notUnique, unproven; public var name: String }`
- `public struct PuzzleCertificate: Sendable, Equatable, Codable { public let verdict: PuzzleVerdict; public let explored: Int; public let shortest: Int?; public let solvedStates: Int; public let shortestPaths: Int; public let maxBranching: Int; public var isValid: Bool }`
- `public struct PuzzleLevel<Rules: PuzzleRules>: Sendable, Equatable { public static var schema: Int { 1 }; public let rulesID: String; public let start: Rules.State; public let solution: [Rules.Move]; public let uniqueness: PuzzleUniqueness; public let budget: FairnessBudget; public let seed: UInt64?; public let identity: String; public init(rulesID: String, start: Rules.State, solution: [Rules.Move], uniqueness: PuzzleUniqueness = .none, budget: FairnessBudget, seed: UInt64? = nil) throws; public func encoded() -> Data; public static func decode(_ data: Data) throws -> PuzzleLevel }`
- `public enum PuzzleValidator { public static func certify<R: PuzzleRules>(_ level: PuzzleLevel<R>, rules: R) -> PuzzleCertificate }`
- `public struct PuzzleGeneration<Rules: PuzzleRules>: Sendable { public let level: PuzzleLevel<Rules>; public let certificate: PuzzleCertificate; public let attempts: Int; public let rejections: [String: Int] }`
- `public enum PuzzleGenerator { public static let maxAttempts = 10_000; public static func generate<R: PuzzleRules>(seed: UInt64, attempts: Int, rules: R, budget: FairnessBudget, uniqueness: PuzzleUniqueness, make: (inout SeededRandom) -> (start: R.State, solution: [R.Move])?) throws -> PuzzleGeneration<R> }`
